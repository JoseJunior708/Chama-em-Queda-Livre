from datetime import date

from sqlalchemy import (
    Boolean,
    Column,
    Date,
    Float,
    ForeignKey,
    Integer,
    String,
    Table,
    Text,
    UniqueConstraint,
)
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.database import Base


# RelaÃ§Ã£o muitos-para-muitos entre experiÃªncias e combustÃ­veis.
experiment_fuels = Table(
    "experiment_fuels",
    Base.metadata,

    Column(
        "experiment_id",
        ForeignKey("experiments.id", ondelete="CASCADE"),
        primary_key=True,
    ),

    Column(
        "fuel_id",
        ForeignKey("fuels.id", ondelete="CASCADE"),
        primary_key=True,
    ),
)


class Experiment(Base):
    __tablename__ = "experiments"

    id: Mapped[int] = mapped_column(
        Integer,
        primary_key=True,
        index=True,
    )

    mission_code: Mapped[str] = mapped_column(
        String(50),
        unique=True,
        index=True,
        nullable=False,
    )

    title: Mapped[str] = mapped_column(
        String(200),
        nullable=False,
    )

    experiment_date: Mapped[date | None] = mapped_column(
        Date,
        nullable=True,
    )

    summary: Mapped[str] = mapped_column(
        Text,
        nullable=False,
    )

    gravity_level: Mapped[str] = mapped_column(
        String(100),
        nullable=False,
    )

    source_url: Mapped[str | None] = mapped_column(
        String(500),
        nullable=True,
    )

    data_status: Mapped[str] = mapped_column(
        String(20),
        default="demo",
        nullable=False,
    )

    # Uma experiÃªncia pode utilizar vÃ¡rios combustÃ­veis.
    fuels: Mapped[list["Fuel"]] = relationship(
        secondary=experiment_fuels,
        back_populates="experiments",
    )

    # Uma experiÃªncia pode ter vÃ¡rias condiÃ§Ãµes ambientais.
    environments: Mapped[list["Environment"]] = relationship(
        back_populates="experiment",
        cascade="all, delete-orphan",
    )

    # Uma experiÃªncia pode ter vÃ¡rias mediÃ§Ãµes.
    telemetry: Mapped[list["TelemetryData"]] = relationship(
        back_populates="experiment",
        cascade="all, delete-orphan",
    )


class Fuel(Base):
    __tablename__ = "fuels"

    id: Mapped[int] = mapped_column(
        Integer,
        primary_key=True,
        index=True,
    )

    fuel_type: Mapped[str] = mapped_column(
        String(100),
        nullable=False,
    )

    chemical_composition: Mapped[str | None] = mapped_column(
        String(200),
        nullable=True,
    )

    physical_state: Mapped[str] = mapped_column(
        String(30),
        nullable=False,
    )

    experiments: Mapped[list["Experiment"]] = relationship(
        secondary=experiment_fuels,
        back_populates="fuels",
    )

    __table_args__ = (
        UniqueConstraint(
            "fuel_type",
            "chemical_composition",
            "physical_state",
            name="uq_fuel_definition",
        ),
    )


class Environment(Base):
    __tablename__ = "environments"

    id: Mapped[int] = mapped_column(
        Integer,
        primary_key=True,
        index=True,
    )

    experiment_id: Mapped[int] = mapped_column(
        ForeignKey("experiments.id", ondelete="CASCADE"),
        nullable=False,
        index=True,
    )

    pressure_kpa: Mapped[float | None] = mapped_column(
        Float,
        nullable=True,
    )

    oxygen_level_pct: Mapped[float | None] = mapped_column(
        Float,
        nullable=True,
    )

    diluent_gas: Mapped[str | None] = mapped_column(
        String(100),
        nullable=True,
    )

    initial_temperature_k: Mapped[float | None] = mapped_column(
        Float,
        nullable=True,
    )

    notes: Mapped[str | None] = mapped_column(
        Text,
        nullable=True,
    )

    experiment: Mapped["Experiment"] = relationship(
        back_populates="environments",
    )


class TelemetryData(Base):
    __tablename__ = "telemetry_data"

    id: Mapped[int] = mapped_column(
        Integer,
        primary_key=True,
        index=True,
    )

    experiment_id: Mapped[int] = mapped_column(
        ForeignKey("experiments.id", ondelete="CASCADE"),
        nullable=False,
        index=True,
    )

    environment_id: Mapped[int | None] = mapped_column(
        ForeignKey("environments.id", ondelete="SET NULL"),
        nullable=True,
    )

    time_s: Mapped[float] = mapped_column(
        Float,
        nullable=False,
    )

    flame_temperature_k: Mapped[float | None] = mapped_column(
        Float,
        nullable=True,
    )

    burning_rate_mm_s: Mapped[float | None] = mapped_column(
        Float,
        nullable=True,
    )

    soot_emission: Mapped[float | None] = mapped_column(
        Float,
        nullable=True,
    )

    is_synthetic: Mapped[bool] = mapped_column(
        Boolean,
        default=True,
        nullable=False,
    )

    experiment: Mapped["Experiment"] = relationship(
        back_populates="telemetry",
    )

    environment: Mapped["Environment | None"] = relationship()

import pandas as pd

from sqlalchemy import select

from app.database import SessionLocal
from app.models import TelemetryData


def load_telemetry_csv(file_path):
    df = pd.read_csv(file_path)

    required = [
        "experiment_id",
        "time_s",
        "flame_temperature_k",
        "burning_rate_mm_s",
        "soot_emission"
    ]

    missing = [
        column for column in required
        if column not in df.columns
    ]

    if missing:
        raise ValueError(
            f"Colunas obrigatÃ³rias em falta: {missing}"
        )

    inserted = 0
    skipped = 0

    with SessionLocal() as session:
        try:
            for _, row in df.iterrows():
                if pd.isna(row["experiment_id"]) or pd.isna(row["time_s"]):
                    skipped += 1
                    continue

                experiment_id = int(row["experiment_id"])
                time_s = float(row["time_s"])

                existing = session.scalar(
                    select(TelemetryData).where(
                        TelemetryData.experiment_id == experiment_id,
                        TelemetryData.time_s == time_s
                    )
                )

                if existing:
                    skipped += 1
                    continue

                record = TelemetryData(
                    experiment_id=experiment_id,
                    time_s=time_s,
                    flame_temperature_k=(
                        None if pd.isna(row["flame_temperature_k"])
                        else float(row["flame_temperature_k"])
                    ),
                    burning_rate_mm_s=(
                        None if pd.isna(row["burning_rate_mm_s"])
                        else float(row["burning_rate_mm_s"])
                    ),
                    soot_emission=(
                        None if pd.isna(row["soot_emission"])
                        else float(row["soot_emission"])
                    ),
                    is_synthetic=(
                        bool(row["is_synthetic"])
                        if "is_synthetic" in df.columns
                        and not pd.isna(row["is_synthetic"])
                        else False
                    )
                )

                session.add(record)
                inserted += 1

            session.commit()

        except Exception:
            session.rollback()
            raise

    return {
        "inserted": inserted,
        "skipped": skipped
    }
